   106→   106→// ─────────────────────────────────────────────────────────────────────────────
   107→   107→// MainScreen – the primary browser screen that orchestrates the entire app.
   108→   108→// ─────────────────────────────────────────────────────────────────────────────
   109→   109→
   110→   110→/**
   111→   111→ * The top-level browser screen that hosts the [Omnibox], WebView content,
   112→   112→ * [TabStrip], and all overlay bottom-sheets / dialogs.
   113→   113→ *
   114→   114→ * This composable is the primary entry point after onboarding and is
   115→   115→ * responsible for orchestrating the [MainViewModel] (tab & workspace
   116→   116→ * management) with the [BrowserViewModel] (WebView rendering).
   117→   117→ *
   118→   118→ * @param initialUrl          An optional URL from a share-intent or deep link.
   119→   119→ * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   120→   120→ * @param onNavigateToSettings  Callback to navigate to the settings screen.
   121→   121→ */
   122→   122→@OptIn(ExperimentalMaterial3Api::class)
   123→   123→@SuppressLint("SetJavaScriptEnabled")
   124→   124→@Composable
   125→   125→fun MainScreen(
   126→   126→    initialUrl: String? = null,
   127→   127→    onNavigateToBookmarks: () -> Unit = {},
   128→   128→    onNavigateToSettings: () -> Unit = {},
   129→   129→    mainViewModel: MainViewModel = hiltViewModel(),
   130→   130→    browserViewModel: BrowserViewModel = hiltViewModel(),
   131→   131→) {
   132→   132→    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   133→   133→    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   134→   134→    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   135→   135→    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   136→   136→
   137→   137→    val coroutineScope = rememberCoroutineScope()
   138→   138→    val snackbarHostState = remember { SnackbarHostState() }
   139→   139→
   140→   140→    // ── Handle initial URL from share intent / deep link ────────────
   141→   141→    val initialUrlHandled = remember { mutableStateOf(false) }
   142→   142→    LaunchedEffect(initialUrl) {
   143→   143→        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   144→   144→            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   145→   145→            initialUrlHandled.value = true
   146→   146→        }
   147→   147→    }
   148→   148→
   149→   149→    // ── Track active tab changes and attach to browser ──────────────
   150→   150→    val activeTab = uiState.activeTab
   151→   151→    LaunchedEffect(activeTab?.id) {
   152→   152→        if (activeTab != null && !activeTab.isHibernated) {
   153→   153→            browserViewModel.attachTab(activeTab.id, activeTab)
   154→   154→        }
   155→   155→    }
   156→   156→
   157→   157→    // Refresh bookmark status when the active tab URL changes
   158→   158→    LaunchedEffect(activeTab?.url) {
   159→   159→        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   160→   160→    }
   161→   161→
   162→   162→    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   163→   163→    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   164→   164→
   165→   165→    // ── Drawer state for workspace rail (wide screens / all screens) ─
   166→   166→    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   167→   167→
   168→   168→    // ── Menu expansion state ────────────────────────────────────────
   169→   169→    var showOverflowMenu by remember { mutableStateOf(false) }
   170→   170→
   171→   171→    // ═══════════════════════════════════════════════════════════════════
   172→   172→    // Back-press handling: sheets > WebView.goBack > close tab
   173→   173→    // ═══════════════════════════════════════════════════════════════════
   174→   174→    BackHandler(enabled = true) {
   175→   175→        when {
   176→   176→            uiState.showQuickSwitcher ||
   177→   177→            uiState.showWorkspaceSwitcher ||
   178→   178→            uiState.showAddTabSheet ||
   179→   179→            uiState.showRecentlyClosed -> {
   180→   180→                mainViewModel.onEvent(MainEvent.DismissAll)
   181→   181→            }
   182→   182→            uiState.showTabOverview -> {
   183→   183→                mainViewModel.onEvent(MainEvent.DismissAll)
   184→   184→            }
   185→   185→            showOverflowMenu -> {
   186→   186→                showOverflowMenu = false
   187→   187→            }
   188→   188→            browserState.canGoBack -> {
   189→   189→                browserViewModel.goBack()
   190→   190→            }
   191→   191→            activeTab != null -> {
   192→   192→                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   193→   193→            }
   194→   194→        }
   195→   195→    }
   196→   196→
   197→   197→    // ═══════════════════════════════════════════════════════════════════
   198→   198→    // Fullscreen video overlay
   199→   199→    // ═══════════════════════════════════════════════════════════════════
   200→   200→    val fullscreenView = customView
   201→   201→    if (fullscreenView != null) {
   202→   202→        FullscreenVideoOverlay(
   203→   203→            customView = fullscreenView,
   204→   204→            onExitFullscreen = { browserViewModel.exitFullscreen() }
   205→   205→        )
   206→   206→        // Don't render the rest of the UI while in fullscreen
   207→   207→        return
   208→   208→    }
   209→   209→
   210→   210→    // ═══════════════════════════════════════════════════════════════════
   211→   211→    // Quick Switcher overlay
   212→   212→    // ═══════════════════════════════════════════════════════════════════
   213→   213→    if (uiState.showQuickSwitcher) {
   214→   214→        QuickSwitcherOverlay(
   215→   215→            query = uiState.quickSwitcherQuery,
   216→   216→            tabs = uiState.filteredTabs,
   217→   217→            activeTabId = uiState.activeTabId,
   218→   218→            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   219→   219→            onTabSelected = {
   220→   220→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   221→   221→                mainViewModel.onEvent(MainEvent.DismissAll)
   222→   222→            },
   223→   223→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   224→   224→        )
   225→   225→        return
   226→   226→    }
   227→   227→
   228→   228→    // ═══════════════════════════════════════════════════════════════════
   229→   229→    // Tab Overview (fullscreen grid)
   230→   230→    // ═══════════════════════════════════════════════════════════════════
   231→   231→    if (uiState.showTabOverview) {
   232→   232→        TabOverviewScreen(
   233→   233→            viewModel = hiltViewModel(),
   234→   234→            onTabSelected = {
   235→   235→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   236→   236→                mainViewModel.onEvent(MainEvent.DismissAll)
   237→   237→            },
   238→   238→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   239→   239→            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   240→   240→        )
   241→   241→        return
   242→   242→    }
   243→   243→
   244→   244→    // ═══════════════════════════════════════════════════════════════════
   245→   245→    // Main browser layout
   246→   246→    // ═══════════════════════════════════════════════════════════════════
   247→   247→    ModalNavigationDrawer(
   248→   248→        drawerState = drawerState,
   249→   249→        drawerContent = {
   250→   250→            WorkspaceDrawerContent(
   251→   251→                workspaces = uiState.workspaces,
   252→   252→                activeWorkspaceId = uiState.activeWorkspaceId,
   253→   253→                onWorkspaceSelected = { wsId ->
   254→   254→                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   255→   255→                    coroutineScope.launch { drawerState.close() }
   256→   256→                },
   257→   257→                onNavigateToBookmarks = {
   258→   258→                    coroutineScope.launch { drawerState.close() }
   259→   259→                    onNavigateToBookmarks()
   260→   260→                },
   261→   261→                onNavigateToSettings = {
   262→   262→                    coroutineScope.launch { drawerState.close() }
   263→   263→                    onNavigateToSettings()
   264→   264→                },
   265→   265→                onNavigateToHistory = {
   266→   266→                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   267→   267→                    coroutineScope.launch { drawerState.close() }
   268→   268→                }
   269→   269→            )
   270→   270→        }
   271→   271→    ) {
   272→   272→        Box(
   273→   273→            modifier = Modifier
   274→   274→                .fillMaxSize()
   275→   275→        ) {
   276→   276→            Scaffold(
   277→   277→                modifier = Modifier
   278→   278→                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   279→   279→                    .navigationBarsPadding(),
   280→   280→                topBar = {
   281→   281→                    MainTopBar(
   282→   282→                        uiState = uiState,
   283→   283→                        browserState = browserState,
   284→   284→                        scrollBehavior = scrollBehavior,
   285→   285→                        onUrlSubmit = { input ->
   286→   286→                            val result = UrlNormalizer.normalize(
   287→   287→                                input,
   288→   288→                                uiState.settings.searchEngineUrl
   289→   289→                            )
   290→   290→                            browserViewModel.loadUrl(result.url)
   291→   291→                        },
   292→   292→                        onBack = { browserViewModel.goBack() },
   293→   293→                        onForward = { browserViewModel.goForward() },
   294→   294→                        onRefresh = {
   295→   295→                            if (browserState.isLoading) browserViewModel.stopLoading()
   296→   296→                            else browserViewModel.reload()
   297→   297→                        },
   298→   298→                        onBookmarkToggle = {
   299→   299→                            val tab = activeTab ?: return@MainTopBar
   300→   300→                            mainViewModel.onEvent(
   301→   301→                                MainEvent.ToggleBookmark(
   302→   302→                                    url = tab.url,
   303→   303→                                    title = tab.title,
   304→   304→                                    faviconUrl = tab.faviconUrl
   305→   305→                                )
   306→   306→                            )
   307→   307→                        },
   308→   308→                        onMenuClick = { showOverflowMenu = true },
   309→   309→                        onDrawerClick = {
   310→   310→                            coroutineScope.launch { drawerState.open() }
   311→   311→                        }
   312→   312→                    )
   313→   313→                },
   314→   314→                bottomBar = {
   315→   315→                    if (uiState.tabs.isNotEmpty()) {
   316→   316→                        TabStrip(
   317→   317→                            tabs = uiState.tabs,
   318→   318→                            activeTabId = uiState.activeTabId,
   319→   319→                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   320→   320→                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   321→   321→                            onTabLongClick = {
   322→   322→                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   323→   323→                            },
   324→   324→                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   325→   325→                        )
   326→   326→                    }
   327→   327→                },
   328→   328→                floatingActionButton = {
   329→   329→                    ExtendedNewTabFab(
   330→   330→                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   331→   331→                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   332→   332→                    )
   333→   333→                },
   334→   334→                snackbarHost = { SnackbarHost(snackbarHostState) },
   335→   335→                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   336→   336→            ) { paddingValues ->
   337→   337→
   338→   338→                // ── WebView content area ────────────────────────────
   339→   339→                Box(
   340→   340→                    modifier = Modifier
   341→   341→                        .fillMaxSize()
   342→   342→                        .padding(paddingValues)
   343→   343→                ) {
   344→   344→                    val currentWebView = webView
   345→   345→                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   346→   346→                        AndroidView(
   347→   347→                            factory = { currentWebView },
   348→   348→                            modifier = Modifier.fillMaxSize()
   349→   349→                        )
   350→   350→                    } else if (activeTab?.isHibernated == true) {
   351→   351→                        HibernatedTabPlaceholder(
   352→   352→                            tab = activeTab,
   353→   353→                            onWake = {
   354→   354→                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   355→   355→                            }
   356→   356→                        )
   357→   357→                    } else {
   358→   358→                        HomePlaceholder(
   359→   359→                            onSearch = { query ->
   360→   360→                                val result = UrlNormalizer.normalize(
   361→   361→                                    query,
   362→   362→                                    uiState.settings.searchEngineUrl
   363→   363→                                )
   364→   364→                                if (activeTab != null) {
   365→   365→                                    browserViewModel.loadUrl(result.url)
   366→   366→                                } else {
   367→   367→                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   368→   368→                                }
   369→   369→                            },
   370→   370→                            onNavigateToBookmarks = onNavigateToBookmarks
   371→   371→                        )
   372→   372→                    }
   373→   373→
   374→   374→                }
   375→   375→            }
   376→   376→        }
   377→   377→    }
   378→   378→
   379→   379→    // ═══════════════════════════════════════════════════════════════════
   380→   380→    // Overflow dropdown menu
   381→   381→    // ═══════════════════════════════════════════════════════════════════
   382→   382→    if (showOverflowMenu) {
   383→   383→        OverflowMenuDialog(
   384→   384→            onDismiss = { showOverflowMenu = false },
   385→   385→            onTabOverview = {
   386→   386→                showOverflowMenu = false
   387→   387→                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   388→   388→            },
   389→   389→            onNewTab = {
   390→   390→                showOverflowMenu = false
   391→   391→                mainViewModel.onEvent(MainEvent.ShowAddTab)
   392→   392→            },
   393→   393→            onBookmarks = {
   394→   394→                showOverflowMenu = false
   395→   395→                onNavigateToBookmarks()
   396→   396→            },
   397→   397→            onHistory = {
   398→   398→                showOverflowMenu = false
   399→   399→                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   400→   400→            },
   401→   401→            onSettings = {
   402→   402→                showOverflowMenu = false
   403→   403→                onNavigateToSettings()
   404→   404→            }
   405→   405→        )
   406→   406→    }
   407→   407→
   408→   408→    // ═══════════════════════════════════════════════════════════════════
   409→   409→    // Bottom sheets
   410→   410→    // ═══════════════════════════════════════════════════════════════════
   411→   411→    if (uiState.showWorkspaceSwitcher) {
   412→   412→        WorkspaceSwitcherSheet(
   413→   413→            viewModel = hiltViewModel(),
   414→   414→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   415→   415→        )
   416→   416→    }
   417→   417→
   418→   418→    if (uiState.showAddTabSheet) {
   419→   419→        AddTabSheet(
   420→   420→            viewModel = hiltViewModel(),
   421→   421→            onTabCreated = { params ->
   422→   422→                mainViewModel.onEvent(MainEvent.AddTab(params.url, params.customName))
   423→   423→                mainViewModel.onEvent(MainEvent.DismissAll)
   424→   424→            },
   425→   425→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   426→   426→        )
   427→   427→    }
   428→   428→
   429→   429→    if (uiState.showRecentlyClosed) {
   430→   430→        RecentlyClosedSheet(
   431→   431→            closedTabs = emptyList(),
   432→   432→            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   433→   433→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   434→   434→            onDismissItem = {},
   435→   435→            onClearAll = {}
   436→   436→        )
   437→   437→    }
   438→   438→}
   439→   439→
   440→   440→// ─────────────────────────────────────────────────────────────────────────────
   441→   441→// TopAppBar with Omnibox

   444→   444→@OptIn(ExperimentalMaterial3Api::class)
   445→   445→@Composable
   446→   446→private fun MainTopBar(
   447→   447→    uiState: MainUiState,
   448→   448→    browserState: BrowserUiState,
   449→   449→    scrollBehavior: TopAppBarScrollBehavior,
   450→   450→    onUrlSubmit: (String) -> Unit,
   451→   451→    onBack: () -> Unit,
   452→   452→    onForward: () -> Unit,
   453→   453→    onRefresh: () -> Unit,
   454→   454→    onBookmarkToggle: () -> Unit,
   455→   455→    onMenuClick: () -> Unit,
   456→   456→    onDrawerClick: () -> Unit,
