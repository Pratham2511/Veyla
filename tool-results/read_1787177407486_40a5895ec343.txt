   575→@OptIn(ExperimentalMaterial3Api::class)
   576→@Composable
   577→private fun SearchEngineDropdown(
   578→    engines: List<Pair<String, String>>,
   579→    currentEngineName: String,
   580→    onEngineSelected: (String) -> Unit,
   581→) {
   582→    var expanded by remember { mutableStateOf(false) }
   583→
   584→    Column(modifier = Modifier.padding(horizontal = 16.dp, vertical = 4.dp)) {
   585→        Text(
   586→            text = "Default search engine",
   587→            style = MaterialTheme.typography.bodyMedium,
   588→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   589→            modifier = Modifier.padding(bottom = 8.dp)
   590→        )
   591→
   592→        ExposedDropdownMenuBox(
   593→            expanded = expanded,
   594→            onExpandedChange = { expanded = it }
   595→        ) {
   596→            OutlinedTextField(
   597→                value = currentEngineName,
   598→                onValueChange = {},
   599→                readOnly = true,
   600→                label = { Text("Search engine") },
   601→                trailingIcon = {
   602→                    ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
   603→                },
   604→                leadingIcon = {
   605→                    Icon(Icons.Default.Search, contentDescription = null)
   606→                },
   607→                modifier = Modifier
   608→                    .menuAnchor(MenuAnchorType.PrimaryNotEditable)
   609→                    .fillMaxWidth()
   610→            )
   611→
   612→            ExposedDropdownMenu(
   613→                expanded = expanded,
   614→                onDismissRequest = { expanded = false }
   615→            ) {
   616→                engines.forEach { (name, url) ->
   617→                    androidx.compose.material3.DropdownMenuItem(
   618→                        text = { Text(name) },
   619→                        onClick = {
   620→                            onEngineSelected(url)
   621→                            expanded = false
   622→                        },
   623→                        trailingIcon = {
   624→                            if (name == currentEngineName) {
   625→                                Text(
   626→                                    "\u2713", // checkmark
   627→                                    color = MaterialTheme.colorScheme.primary
   628→                                )
   629→                            }
   630→                        }
   631→                    )
   632→                }
   633→            }
   634→        }
   635→    }
   636→}

   638→@Composable
