   105→// ─────────────────────────────────────────────────────────────────────────────
   106→// MainScreen – the primary browser screen that orchestrates the entire app.
   107→// ─────────────────────────────────────────────────────────────────────────────
   108→
   109→/**
   110→ * The top-level browser screen that hosts the [Omnibox], WebView content,
   111→ * [TabStrip], and all overlay bottom-sheets / dialogs.
   112→ *
   113→ * This composable is the primary entry point after onboarding and is
   114→ * responsible for orchestrating the [MainViewModel] (tab & workspace
   115→ * management) with the [BrowserViewModel] (WebView rendering).
   116→ *
   117→ * @param initialUrl          An optional URL from a share-intent or deep link.
   118→ * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   119→ * @param onNavigateToSettings  Callback to navigate to the settings screen.
   120→ */
   121→@OptIn(ExperimentalMaterial3Api::class)
   122→@SuppressLint("SetJavaScriptEnabled")
   123→@Composable
   124→fun MainScreen(
   125→    initialUrl: String? = null,
   126→    onNavigateToBookmarks: () -> Unit = {},
   127→    onNavigateToSettings: () -> Unit = {},
   128→    mainViewModel: MainViewModel = hiltViewModel(),
   129→    browserViewModel: BrowserViewModel = hiltViewModel(),
   130→) {
   131→    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   132→    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   133→    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   134→    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   135→
   136→    val coroutineScope = rememberCoroutineScope()
   137→    val snackbarHostState = remember { SnackbarHostState() }
   138→
   139→    // ── Handle initial URL from share intent / deep link ────────────
   140→    val initialUrlHandled = remember { mutableStateOf(false) }
   141→    LaunchedEffect(initialUrl) {
   142→        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   143→            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   144→            initialUrlHandled.value = true
   145→        }
   146→    }
   147→
   148→    // ── Track active tab changes and attach to browser ──────────────
   149→    val activeTab = uiState.activeTab
   150→    LaunchedEffect(activeTab?.id) {
   151→        if (activeTab != null && !activeTab.isHibernated) {
   152→            browserViewModel.attachTab(activeTab.id, activeTab)
   153→        }
   154→    }
   155→
   156→    // Refresh bookmark status when the active tab URL changes
   157→    LaunchedEffect(activeTab?.url) {
   158→        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   159→    }
   160→
   161→    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   162→    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   163→
   164→    // ── Drawer state for workspace rail (wide screens / all screens) ─
   165→    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   166→
   167→    // ── Menu expansion state ────────────────────────────────────────
   168→    var showOverflowMenu by remember { mutableStateOf(false) }
   169→
   170→    // ═══════════════════════════════════════════════════════════════════
   171→    // Back-press handling: sheets > WebView.goBack > close tab
   172→    // ═══════════════════════════════════════════════════════════════════
   173→    BackHandler(enabled = true) {
   174→        when {
   175→            uiState.showQuickSwitcher ||
   176→            uiState.showWorkspaceSwitcher ||
   177→            uiState.showAddTabSheet ||
   178→            uiState.showRecentlyClosed -> {
   179→                mainViewModel.onEvent(MainEvent.DismissAll)
   180→            }
   181→            uiState.showTabOverview -> {
   182→                mainViewModel.onEvent(MainEvent.DismissAll)
   183→            }
   184→            showOverflowMenu -> {
   185→                showOverflowMenu = false
   186→            }
   187→            browserState.canGoBack -> {
   188→                browserViewModel.goBack()
   189→            }
   190→            activeTab != null -> {
   191→                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   192→            }
   193→        }
   194→    }
   195→
   196→    // ═══════════════════════════════════════════════════════════════════
   197→    // Fullscreen video overlay
   198→    // ═══════════════════════════════════════════════════════════════════
   199→    if (customView != null) {
   200→        FullscreenVideoOverlay(
   201→            customView = customView,
   202→            onExitFullscreen = { browserViewModel.exitFullscreen() }
   203→        )
   204→        // Don't render the rest of the UI while in fullscreen
   205→        return
   206→    }
   207→
   208→    // ═══════════════════════════════════════════════════════════════════
   209→    // Quick Switcher overlay
   210→    // ═══════════════════════════════════════════════════════════════════
   211→    if (uiState.showQuickSwitcher) {
   212→        QuickSwitcherOverlay(
   213→            query = uiState.quickSwitcherQuery,
   214→            tabs = uiState.filteredTabs,
   215→            activeTabId = uiState.activeTabId,
   216→            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   217→            onTabSelected = {
   218→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   219→                mainViewModel.onEvent(MainEvent.DismissAll)
   220→            },
   221→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   222→        )
   223→        return
   224→    }
   225→
   226→    // ═══════════════════════════════════════════════════════════════════
   227→    // Tab Overview (fullscreen grid)
   228→    // ═══════════════════════════════════════════════════════════════════
   229→    if (uiState.showTabOverview) {
   230→        TabOverviewScreen(
   231→            viewModel = hiltViewModel(),
   232→            onTabSelected = {
   233→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   234→                mainViewModel.onEvent(MainEvent.DismissAll)
   235→            },
   236→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   237→            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   238→        )
   239→        return
   240→    }
   241→
   242→    // ═══════════════════════════════════════════════════════════════════
   243→    // Main browser layout
   244→    // ═══════════════════════════════════════════════════════════════════
   245→    ModalNavigationDrawer(
   246→        drawerState = drawerState,
   247→        drawerContent = {
   248→            WorkspaceDrawerContent(
   249→                workspaces = uiState.workspaces,
   250→                activeWorkspaceId = uiState.activeWorkspaceId,
   251→                onWorkspaceSelected = { wsId ->
   252→                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   253→                    coroutineScope.launch { drawerState.close() }
   254→                },
   255→                onNavigateToBookmarks = {
   256→                    coroutineScope.launch { drawerState.close() }
   257→                    onNavigateToBookmarks()
   258→                },
   259→                onNavigateToSettings = {
   260→                    coroutineScope.launch { drawerState.close() }
   261→                    onNavigateToSettings()
   262→                },
   263→                onNavigateToHistory = {
   264→                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   265→                    coroutineScope.launch { drawerState.close() }
   266→                }
   267→            )
   268→        }
   269→    ) {
   270→        Box(
   271→            modifier = Modifier
   272→                .fillMaxSize()
   273→        ) {
   274→            Scaffold(
   275→                modifier = Modifier
   276→                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   277→                    .navigationBarsPadding(),
   278→                topBar = {
   279→                    MainTopBar(
   280→                        uiState = uiState,
   281→                        browserState = browserState,
   282→                        scrollBehavior = scrollBehavior,
   283→                        onUrlSubmit = { input ->
   284→                            val result = UrlNormalizer.normalize(
   285→                                input,
   286→                                uiState.settings.searchEngineUrl
   287→                            )
   288→                            browserViewModel.loadUrl(result.url)
   289→                        },
   290→                        onBack = { browserViewModel.goBack() },
   291→                        onForward = { browserViewModel.goForward() },
   292→                        onRefresh = {
   293→                            if (browserState.isLoading) browserViewModel.stopLoading()
   294→                            else browserViewModel.reload()
   295→                        },
   296→                        onBookmarkToggle = {
   297→                            val tab = activeTab ?: return@MainTopBar
   298→                            mainViewModel.onEvent(
   299→                                MainEvent.ToggleBookmark(
   300→                                    url = tab.url,
   301→                                    title = tab.title,
   302→                                    faviconUrl = tab.faviconUrl
   303→                                )
   304→                            )
   305→                        },
   306→                        onMenuClick = { showOverflowMenu = true },
   307→                        onDrawerClick = {
   308→                            coroutineScope.launch { drawerState.open() }
   309→                        }
   310→                    )
   311→                },
   312→                bottomBar = {
   313→                    if (uiState.tabs.isNotEmpty()) {
   314→                        TabStrip(
   315→                            tabs = uiState.tabs,
   316→                            activeTabId = uiState.activeTabId,
   317→                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   318→                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   319→                            onTabLongClick = {
   320→                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   321→                            },
   322→                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   323→                        )
   324→                    }
   325→                },
   326→                floatingActionButton = {
   327→                    ExtendedNewTabFab(
   328→                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   329→                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   330→                    )
   331→                },
   332→                snackbarHost = { SnackbarHost(snackbarHostState) },
   333→                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   334→            ) { paddingValues ->
   335→
   336→                // ── WebView content area ────────────────────────────
   337→                Box(
   338→                    modifier = Modifier
   339→                        .fillMaxSize()
   340→                        .padding(paddingValues)
   341→                ) {
   342→                    val currentWebView = webView
   343→                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   344→                        AndroidView(
   345→                            factory = { currentWebView },
   346→                            modifier = Modifier.fillMaxSize()
   347→                        )
   348→                    } else if (activeTab?.isHibernated == true) {
   349→                        HibernatedTabPlaceholder(
   350→                            tab = activeTab,
   351→                            onWake = {
   352→                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   353→                            }
   354→                        )
   355→                    } else {
   356→                        HomePlaceholder(
   357→                            onSearch = { query ->
   358→                                val result = UrlNormalizer.normalize(
   359→                                    query,
   360→                                    uiState.settings.searchEngineUrl
   361→                                )
   362→                                if (activeTab != null) {
   363→                                    browserViewModel.loadUrl(result.url)
   364→                                } else {
   365→                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   366→                                }
   367→                            },
   368→                            onNavigateToBookmarks = onNavigateToBookmarks
   369→                        )
   370→                    }
   371→
   372→                }
   373→            }
   374→        }
   375→    }
   376→
   377→    // ═══════════════════════════════════════════════════════════════════
   378→    // Overflow dropdown menu
   379→    // ═══════════════════════════════════════════════════════════════════
   380→    if (showOverflowMenu) {
   381→        OverflowMenuDialog(
   382→            onDismiss = { showOverflowMenu = false },
   383→            onTabOverview = {
   384→                showOverflowMenu = false
   385→                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   386→            },
   387→            onNewTab = {
   388→                showOverflowMenu = false
   389→                mainViewModel.onEvent(MainEvent.ShowAddTab)
   390→            },
   391→            onBookmarks = {
   392→                showOverflowMenu = false
   393→                onNavigateToBookmarks()
   394→            },
   395→            onHistory = {
   396→                showOverflowMenu = false
   397→                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   398→            },
   399→            onSettings = {
   400→                showOverflowMenu = false
   401→                onNavigateToSettings()
   402→            }
   403→        )
   404→    }
   405→
   406→    // ═══════════════════════════════════════════════════════════════════
   407→    // Bottom sheets
   408→    // ═══════════════════════════════════════════════════════════════════
   409→    if (uiState.showWorkspaceSwitcher) {
   410→        WorkspaceSwitcherSheet(
   411→            viewModel = hiltViewModel(),
   412→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   413→        )
   414→    }
   415→
   416→    if (uiState.showAddTabSheet) {
   417→        AddTabSheet(
   418→            viewModel = hiltViewModel(),
   419→            onTabCreated = { params ->
   420→                mainViewModel.onEvent(MainEvent.AddTab(params.url, params.customName))
   421→                mainViewModel.onEvent(MainEvent.DismissAll)
   422→            },
   423→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   424→        )
   425→    }
   426→
   427→    if (uiState.showRecentlyClosed) {
   428→        RecentlyClosedSheet(
   429→            closedTabs = emptyList(),
   430→            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   431→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   432→            onDismissItem = {},
   433→            onClearAll = {}
   434→        )
   435→    }
   436→}

   442→@OptIn(ExperimentalMaterial3Api::class)
   443→@Composable
   444→private fun MainTopBar(
   445→    uiState: MainUiState,
   446→    browserState: BrowserUiState,
   447→    scrollBehavior: TopAppBarScrollBehavior,
   448→    onUrlSubmit: (String) -> Unit,
   449→    onBack: () -> Unit,
   450→    onForward: () -> Unit,
   451→    onRefresh: () -> Unit,
   452→    onBookmarkToggle: () -> Unit,
   453→    onMenuClick: () -> Unit,
   454→    onDrawerClick: () -> Unit,
   455→) {
   456→    Column {
   457→        TopAppBar(
   458→            title = {},
   459→            navigationIcon = {
   460→                IconButton(onClick = onDrawerClick) {
   461→                    Icon(
   462→                        imageVector = Icons.Default.Folder,
   463→                        contentDescription = "Workspaces",
   464→                        tint = MaterialTheme.colorScheme.onSurface
   465→                    )
   466→                }
   467→            },
   468→            actions = {
   469→                val tabCount = uiState.tabs.size
   470→                Surface(
   471→                    shape = RoundedCornerShape(12.dp),
   472→                    color = MaterialTheme.colorScheme.surfaceVariant,
   473→                    modifier = Modifier
   474→                        .padding(end = 4.dp)
   475→                        .clickable { /* Tab count chip – no action here */ }
   476→                ) {
   477→                    Row(
   478→                        verticalAlignment = Alignment.CenterVertically,
   479→                        modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp)
   480→                    ) {
   481→                        Icon(
   482→                            imageVector = Icons.Default.Tab,
   483→                            contentDescription = "Tabs",
   484→                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
   485→                            modifier = Modifier.size(16.dp)
   486→                        )
   487→                        Spacer(Modifier.width(4.dp))
   488→                        Text(
   489→                            text = "$tabCount",
   490→                            style = MaterialTheme.typography.labelSmall,
   491→                            color = MaterialTheme.colorScheme.onSurfaceVariant
   492→                        )
   493→                    }
   494→                }
   495→            },
   496→            colors = TopAppBarDefaults.topAppBarColors(
   497→                containerColor = Color.Transparent
   498→            ),
   499→            scrollBehavior = scrollBehavior,
   500→            modifier = Modifier.statusBarsPadding()
   501→        )
   502→
   503→        // Omnibox below the TopAppBar
   504→        Omnibox(
   505→            url = browserState.currentUrl,
   506→            title = browserState.currentTitle,
   507→            isLoading = browserState.isLoading,
   508→            progress = browserState.loadingProgress,
   509→            sslState = browserState.sslState,
   510→            canGoBack = browserState.canGoBack,
   511→            canGoForward = browserState.canGoForward,
   512→            onUrlSubmit = onUrlSubmit,
   513→            onBack = onBack,
   514→            onForward = onForward,
   515→            onRefresh = onRefresh,
   516→            onBookmarkToggle = onBookmarkToggle,
   517→            isBookmarked = uiState.isBookmarked,
   518→            onMenuClick = onMenuClick,
   519→            modifier = Modifier.padding(horizontal = 4.dp)
   520→        )
   521→    }
   522→}

   528→@Composable
   529→private fun WorkspaceDrawerContent(
   530→    workspaces: List<Workspace>,
   531→    activeWorkspaceId: String?,
   532→    onWorkspaceSelected: (String) -> Unit,
   533→    onNavigateToBookmarks: () -> Unit,
   534→    onNavigateToSettings: () -> Unit,
   535→    onNavigateToHistory: () -> Unit,
   536→) {
   537→    ModalDrawerSheet(
   538→        drawerContainerColor = MaterialTheme.colorScheme.surface
   539→    ) {
   540→        // Header
   541→        Column(
   542→            modifier = Modifier.padding(horizontal = 16.dp, vertical = 12.dp)
   543→        ) {
   544→            Text(
   545→                text = "WebHub",
   546→                style = MaterialTheme.typography.titleLarge,
   547→                fontWeight = FontWeight.Bold,
   548→                color = MaterialTheme.colorScheme.onSurface
   549→            )
   550→            Spacer(Modifier.height(4.dp))
   551→            Text(
   552→                text = "Workspaces",
   553→                style = MaterialTheme.typography.labelMedium,
   554→                color = MaterialTheme.colorScheme.onSurfaceVariant
   555→            )
   556→        }
   557→
   558→        HorizontalDivider(
   559→            modifier = Modifier.padding(vertical = 4.dp),
   560→            color = MaterialTheme.colorScheme.outlineVariant
   561→        )
   562→
   563→        // Workspace list
   564→        LazyColumn(
   565→            modifier = Modifier.weight(1f),
   566→            contentPadding = PaddingValues(vertical = 4.dp)
