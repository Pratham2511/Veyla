   106	// ─────────────────────────────────────────────────────────────────────────────
   107	// MainScreen – the primary browser screen that orchestrates the entire app.
   108	// ─────────────────────────────────────────────────────────────────────────────
   109	
   110	/**
   111	 * The top-level browser screen that hosts the [Omnibox], WebView content,
   112	 * [TabStrip], and all overlay bottom-sheets / dialogs.
   113	 *
   114	 * This composable is the primary entry point after onboarding and is
   115	 * responsible for orchestrating the [MainViewModel] (tab & workspace
   116	 * management) with the [BrowserViewModel] (WebView rendering).
   117	 *
   118	 * @param initialUrl          An optional URL from a share-intent or deep link.
   119	 * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   120	 * @param onNavigateToSettings  Callback to navigate to the settings screen.
   121	 */
   122	@OptIn(ExperimentalMaterial3Api::class)
   123	@SuppressLint("SetJavaScriptEnabled")
   124	@Composable
   125	fun MainScreen(
   126	    initialUrl: String? = null,
   127	    onNavigateToBookmarks: () -> Unit = {},
   128	    onNavigateToSettings: () -> Unit = {},
   129	    mainViewModel: MainViewModel = hiltViewModel(),
   130	    browserViewModel: BrowserViewModel = hiltViewModel(),
   131	) {
   132	    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   133	    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   134	    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   135	    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   136	
   137	    val coroutineScope = rememberCoroutineScope()
   138	    val snackbarHostState = remember { SnackbarHostState() }
   139	
   140	    // ── Handle initial URL from share intent / deep link ────────────
   141	    val initialUrlHandled = remember { mutableStateOf(false) }
   142	    LaunchedEffect(initialUrl) {
   143	        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   144	            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   145	            initialUrlHandled.value = true
   146	        }
   147	    }
   148	
   149	    // ── Track active tab changes and attach to browser ──────────────
   150	    val activeTab = uiState.activeTab
   151	    LaunchedEffect(activeTab?.id) {
   152	        if (activeTab != null && !activeTab.isHibernated) {
   153	            browserViewModel.attachTab(activeTab.id, activeTab)
   154	        }
   155	    }
   156	
   157	    // Refresh bookmark status when the active tab URL changes
   158	    LaunchedEffect(activeTab?.url) {
   159	        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   160	    }
   161	
   162	    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   163	    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   164	
   165	    // ── Drawer state for workspace rail (wide screens / all screens) ─
   166	    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   167	
   168	    // ── Menu expansion state ────────────────────────────────────────
   169	    var showOverflowMenu by remember { mutableStateOf(false) }
   170	
   171	    // ═══════════════════════════════════════════════════════════════════
   172	    // Back-press handling: sheets > WebView.goBack > close tab
   173	    // ═══════════════════════════════════════════════════════════════════
   174	    BackHandler(enabled = true) {
   175	        when {
   176	            uiState.showQuickSwitcher ||
   177	            uiState.showWorkspaceSwitcher ||
   178	            uiState.showAddTabSheet ||
   179	            uiState.showRecentlyClosed -> {
   180	                mainViewModel.onEvent(MainEvent.DismissAll)
   181	            }
   182	            uiState.showTabOverview -> {
   183	                mainViewModel.onEvent(MainEvent.DismissAll)
   184	            }
   185	            showOverflowMenu -> {
   186	                showOverflowMenu = false
   187	            }
   188	            browserState.canGoBack -> {
   189	                browserViewModel.goBack()
   190	            }
   191	            activeTab != null -> {
   192	                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   193	            }
   194	        }
   195	    }
   196	
   197	    // ═══════════════════════════════════════════════════════════════════
   198	    // Fullscreen video overlay
   199	    // ═══════════════════════════════════════════════════════════════════
   200	    val fullscreenView = customView
   201	    if (fullscreenView != null) {
   202	        FullscreenVideoOverlay(
   203	            customView = fullscreenView,
   204	            onExitFullscreen = { browserViewModel.exitFullscreen() }
   205	        )
   206	        // Don't render the rest of the UI while in fullscreen
   207	        return
   208	    }
   209	
   210	    // ═══════════════════════════════════════════════════════════════════
   211	    // Quick Switcher overlay
   212	    // ═══════════════════════════════════════════════════════════════════
   213	    if (uiState.showQuickSwitcher) {
   214	        QuickSwitcherOverlay(
   215	            query = uiState.quickSwitcherQuery,
   216	            tabs = uiState.filteredTabs,
   217	            activeTabId = uiState.activeTabId,
   218	            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   219	            onTabSelected = {
   220	                mainViewModel.onEvent(MainEvent.SelectTab(it))
   221	                mainViewModel.onEvent(MainEvent.DismissAll)
   222	            },
   223	            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   224	        )
   225	        return
   226	    }
   227	
   228	    // ═══════════════════════════════════════════════════════════════════
   229	    // Tab Overview (fullscreen grid)
   230	    // ═══════════════════════════════════════════════════════════════════
   231	    if (uiState.showTabOverview) {
   232	        TabOverviewScreen(
   233	            viewModel = hiltViewModel(),
   234	            onTabSelected = {
   235	                mainViewModel.onEvent(MainEvent.SelectTab(it))
   236	                mainViewModel.onEvent(MainEvent.DismissAll)
   237	            },
   238	            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   239	            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   240	        )
   241	        return
   242	    }
   243	
   244	    // ═══════════════════════════════════════════════════════════════════
   245	    // Main browser layout
   246	    // ═══════════════════════════════════════════════════════════════════
   247	    ModalNavigationDrawer(
   248	        drawerState = drawerState,
   249	        drawerContent = {
   250	            WorkspaceDrawerContent(
   251	                workspaces = uiState.workspaces,
   252	                activeWorkspaceId = uiState.activeWorkspaceId,
   253	                onWorkspaceSelected = { wsId ->
   254	                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   255	                    coroutineScope.launch { drawerState.close() }
   256	                },
   257	                onNavigateToBookmarks = {
   258	                    coroutineScope.launch { drawerState.close() }
   259	                    onNavigateToBookmarks()
   260	                },
   261	                onNavigateToSettings = {
   262	                    coroutineScope.launch { drawerState.close() }
   263	                    onNavigateToSettings()
   264	                },
   265	                onNavigateToHistory = {
   266	                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   267	                    coroutineScope.launch { drawerState.close() }
   268	                }
   269	            )
   270	        }
   271	    ) {
   272	        Box(
   273	            modifier = Modifier
   274	                .fillMaxSize()
   275	        ) {
   276	            Scaffold(
   277	                modifier = Modifier
   278	                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   279	                    .navigationBarsPadding(),
   280	                topBar = {
   281	                    MainTopBar(
   282	                        uiState = uiState,
   283	                        browserState = browserState,
   284	                        scrollBehavior = scrollBehavior,
   285	                        onUrlSubmit = { input ->
   286	                            val result = UrlNormalizer.normalize(
   287	                                input,
   288	                                uiState.settings.searchEngineUrl
   289	                            )
   290	                            browserViewModel.loadUrl(result.url)
   291	                        },
   292	                        onBack = { browserViewModel.goBack() },
   293	                        onForward = { browserViewModel.goForward() },
   294	                        onRefresh = {
   295	                            if (browserState.isLoading) browserViewModel.stopLoading()
   296	                            else browserViewModel.reload()
   297	                        },
   298	                        onBookmarkToggle = {
   299	                            val tab = activeTab ?: return@MainTopBar
   300	                            mainViewModel.onEvent(
   301	                                MainEvent.ToggleBookmark(
   302	                                    url = tab.url,
   303	                                    title = tab.title,
   304	                                    faviconUrl = tab.faviconUrl
   305	                                )
   306	                            )
   307	                        },
   308	                        onMenuClick = { showOverflowMenu = true },
   309	                        onDrawerClick = {
   310	                            coroutineScope.launch { drawerState.open() }
   311	                        }
   312	                    )
   313	                },
   314	                bottomBar = {
   315	                    if (uiState.tabs.isNotEmpty()) {
   316	                        TabStrip(
   317	                            tabs = uiState.tabs,
   318	                            activeTabId = uiState.activeTabId,
   319	                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   320	                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   321	                            onTabLongClick = {
   322	                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   323	                            },
   324	                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   325	                        )
   326	                    }
   327	                },
   328	                floatingActionButton = {
   329	                    ExtendedNewTabFab(
   330	                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   331	                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   332	                    )
   333	                },
   334	                snackbarHost = { SnackbarHost(snackbarHostState) },
   335	                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   336	            ) { paddingValues ->
   337	
   338	                // ── WebView content area ────────────────────────────
   339	                Box(
   340	                    modifier = Modifier
   341	                        .fillMaxSize()
   342	                        .padding(paddingValues)
   343	                ) {
   344	                    val currentWebView = webView
   345	                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   346	                        AndroidView(
   347	                            factory = { currentWebView },
   348	                            modifier = Modifier.fillMaxSize()
   349	                        )
   350	                    } else if (activeTab?.isHibernated == true) {
   351	                        HibernatedTabPlaceholder(
   352	                            tab = activeTab,
   353	                            onWake = {
   354	                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   355	                            }
   356	                        )
   357	                    } else {
   358	                        HomePlaceholder(
   359	                            onSearch = { query ->
   360	                                val result = UrlNormalizer.normalize(
   361	                                    query,
   362	                                    uiState.settings.searchEngineUrl
   363	                                )
   364	                                if (activeTab != null) {
   365	                                    browserViewModel.loadUrl(result.url)
   366	                                } else {
   367	                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   368	                                }
   369	                            },
   370	                            onNavigateToBookmarks = onNavigateToBookmarks
   371	                        )
   372	                    }
   373	
   374	                }
   375	            }
   376	        }
   377	    }
   378	
   379	    // ═══════════════════════════════════════════════════════════════════
   380	    // Overflow dropdown menu
   381	    // ═══════════════════════════════════════════════════════════════════
   382	    if (showOverflowMenu) {
   383	        OverflowMenuDialog(
   384	            onDismiss = { showOverflowMenu = false },
   385	            onTabOverview = {
   386	                showOverflowMenu = false
   387	                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   388	            },
   389	            onNewTab = {
   390	                showOverflowMenu = false
   391	                mainViewModel.onEvent(MainEvent.ShowAddTab)
   392	            },
   393	            onBookmarks = {
   394	                showOverflowMenu = false
   395	                onNavigateToBookmarks()
   396	            },
   397	            onHistory = {
   398	                showOverflowMenu = false
   399	                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   400	            },
   401	            onSettings = {
   402	                showOverflowMenu = false
   403	                onNavigateToSettings()
   404	            }
   405	        )
   406	    }
   407	
   408	    // ═══════════════════════════════════════════════════════════════════
   409	    // Bottom sheets
   410	    // ═══════════════════════════════════════════════════════════════════
   411	    if (uiState.showWorkspaceSwitcher) {
   412	        WorkspaceSwitcherSheet(
   413	            viewModel = hiltViewModel(),
   414	            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   415	        )
   416	    }
   417	
   418	    if (uiState.showAddTabSheet) {
   419	        AddTabSheet(
   420	            viewModel = hiltViewModel(),
   421	            onTabCreated = { params ->
   422	                mainViewModel.onEvent(MainEvent.AddTab(params.url, params.customName))
   423	                mainViewModel.onEvent(MainEvent.DismissAll)
   424	            },
   425	            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   426	        )
   427	    }
   428	
   429	    if (uiState.showRecentlyClosed) {
   430	        RecentlyClosedSheet(
   431	            closedTabs = emptyList(),
   432	            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   433	            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   434	            onDismissItem = {},
   435	            onClearAll = {}
   436	        )
   437	    }
   438	}

   444	@OptIn(ExperimentalMaterial3Api::class)
   445	@Composable
   446	private fun MainTopBar(
   447	    uiState: MainUiState,
   448	    browserState: BrowserUiState,
   449	    scrollBehavior: TopAppBarScrollBehavior,
   450	    onUrlSubmit: (String) -> Unit,
   451	    onBack: () -> Unit,
   452	    onForward: () -> Unit,
   453	    onRefresh: () -> Unit,
   454	    onBookmarkToggle: () -> Unit,
   455	    onMenuClick: () -> Unit,
   456	    onDrawerClick: () -> Unit,
   457	) {
   458	    Column {
   459	        TopAppBar(
   460	            title = {},
   461	            navigationIcon = {
   462	                IconButton(onClick = onDrawerClick) {
   463	                    Icon(
   464	                        imageVector = Icons.Default.Folder,
   465	                        contentDescription = "Workspaces",
   466	                        tint = MaterialTheme.colorScheme.onSurface
   467	                    )
   468	                }
   469	            },
   470	            actions = {
   471	                val tabCount = uiState.tabs.size
   472	                Surface(
   473	                    shape = RoundedCornerShape(12.dp),
   474	                    color = MaterialTheme.colorScheme.surfaceVariant,
   475	                    modifier = Modifier
   476	                        .padding(end = 4.dp)
   477	                        .clickable { /* Tab count chip – no action here */ }
   478	                ) {
   479	                    Row(
   480	                        verticalAlignment = Alignment.CenterVertically,
   481	                        modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp)
   482	                    ) {
   483	                        Icon(
   484	                            imageVector = Icons.Default.Tab,
   485	                            contentDescription = "Tabs",
   486	                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
   487	                            modifier = Modifier.size(16.dp)
   488	                        )
   489	                        Spacer(Modifier.width(4.dp))
   490	                        Text(
   491	                            text = "$tabCount",
   492	                            style = MaterialTheme.typography.labelSmall,
   493	                            color = MaterialTheme.colorScheme.onSurfaceVariant
   494	                        )
   495	                    }
   496	                }
   497	            },
   498	            colors = TopAppBarDefaults.topAppBarColors(
   499	                containerColor = Color.Transparent
   500	            ),
   501	            scrollBehavior = scrollBehavior,
   502	            modifier = Modifier.statusBarsPadding()
   503	        )
   504	
   505	        // Omnibox below the TopAppBar
   506	        Omnibox(
   507	            url = browserState.currentUrl,
   508	            title = browserState.currentTitle,
   509	            isLoading = browserState.isLoading,
   510	            progress = browserState.loadingProgress,
   511	            sslState = browserState.sslState,
   512	            canGoBack = browserState.canGoBack,
   513	            canGoForward = browserState.canGoForward,
   514	            onUrlSubmit = onUrlSubmit,
   515	            onBack = onBack,
   516	            onForward = onForward,
   517	            onRefresh = onRefresh,
   518	            onBookmarkToggle = onBookmarkToggle,
   519	            isBookmarked = uiState.isBookmarked,
   520	            onMenuClick = onMenuClick,
   521	            modifier = Modifier.padding(horizontal = 4.dp)
   522	        )
   523	    }
   524	}

   532	    workspaces: List<Workspace>,
   533	    activeWorkspaceId: String?,
   534	    onWorkspaceSelected: (String) -> Unit,
   535	    onNavigateToBookmarks: () -> Unit,
   536	    onNavigateToSettings: () -> Unit,
   537	    onNavigateToHistory: () -> Unit,
   538	) {
   539	    ModalDrawerSheet(
   540	        drawerContainerColor = MaterialTheme.colorScheme.surface
   541	    ) {
   542	        // Header
   543	        Column(
   544	            modifier = Modifier.padding(horizontal = 16.dp, vertical = 12.dp)
   545	        ) {
   546	            Text(
   547	                text = "WebHub",
   548	                style = MaterialTheme.typography.titleLarge,
   549	                fontWeight = FontWeight.Bold,
   550	                color = MaterialTheme.colorScheme.onSurface
   551	            )
   552	            Spacer(Modifier.height(4.dp))
   553	            Text(
