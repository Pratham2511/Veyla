   107→// ─────────────────────────────────────────────────────────────────────────────
   108→// MainScreen – the primary browser screen that orchestrates the entire app.
   109→// ─────────────────────────────────────────────────────────────────────────────
   110→
   111→/**
   112→ * The top-level browser screen that hosts the [Omnibox], WebView content,
   113→ * [TabStrip], and all overlay bottom-sheets / dialogs.
   114→ *
   115→ * This composable is the primary entry point after onboarding and is
   116→ * responsible for orchestrating the [MainViewModel] (tab & workspace
   117→ * management) with the [BrowserViewModel] (WebView rendering).
   118→ *
   119→ * @param initialUrl          An optional URL from a share-intent or deep link.
   120→ * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   121→ * @param onNavigateToSettings  Callback to navigate to the settings screen.
   122→ */
   123→@OptIn(ExperimentalMaterial3Api::class)
   124→@SuppressLint("SetJavaScriptEnabled")
   125→@Composable
   126→fun MainScreen(
   127→    initialUrl: String? = null,
   128→    onNavigateToBookmarks: () -> Unit = {},
   129→    onNavigateToSettings: () -> Unit = {},
   130→    mainViewModel: MainViewModel = hiltViewModel(),
   131→    browserViewModel: BrowserViewModel = hiltViewModel(),
   132→) {
   133→    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   134→    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   135→    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   136→    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   137→
   138→    val coroutineScope = rememberCoroutineScope()
   139→    val snackbarHostState = remember { SnackbarHostState() }
   140→
   141→    // ── Handle initial URL from share intent / deep link ────────────
   142→    val initialUrlHandled = remember { mutableStateOf(false) }
   143→    LaunchedEffect(initialUrl) {
   144→        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   145→            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   146→            initialUrlHandled.value = true
   147→        }
   148→    }
   149→
   150→    // ── Track active tab changes and attach to browser ──────────────
   151→    val activeTab = uiState.activeTab
   152→    LaunchedEffect(activeTab?.id) {
   153→        if (activeTab != null && !activeTab.isHibernated) {
   154→            browserViewModel.attachTab(activeTab.id, activeTab)
   155→        }
   156→    }
   157→
   158→    // Refresh bookmark status when the active tab URL changes
   159→    LaunchedEffect(activeTab?.url) {
   160→        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   161→    }
   162→
   163→    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   164→    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   165→
   166→    // ── Drawer state for workspace rail (wide screens / all screens) ─
   167→    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   168→
   169→    // ── Menu expansion state ────────────────────────────────────────
   170→    var showOverflowMenu by remember { mutableStateOf(false) }
   171→
   172→    // ═══════════════════════════════════════════════════════════════════
   173→    // Back-press handling: sheets > WebView.goBack > close tab
   174→    // ═══════════════════════════════════════════════════════════════════
   175→    BackHandler(enabled = true) {
   176→        when {
   177→            uiState.showQuickSwitcher ||
   178→            uiState.showWorkspaceSwitcher ||
   179→            uiState.showAddTabSheet ||
   180→            uiState.showRecentlyClosed -> {
   181→                mainViewModel.onEvent(MainEvent.DismissAll)
   182→            }
   183→            uiState.showTabOverview -> {
   184→                mainViewModel.onEvent(MainEvent.DismissAll)
   185→            }
   186→            showOverflowMenu -> {
   187→                showOverflowMenu = false
   188→            }
   189→            browserState.canGoBack -> {
   190→                browserViewModel.goBack()
   191→            }
   192→            activeTab != null -> {
   193→                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   194→            }
   195→        }
   196→    }
   197→
   198→    // ═══════════════════════════════════════════════════════════════════
   199→    // Fullscreen video overlay
   200→    // ═══════════════════════════════════════════════════════════════════
   201→    val fullscreenView = customView
   202→    if (fullscreenView != null) {
   203→        FullscreenVideoOverlay(
   204→            customView = fullscreenView,
   205→            onExitFullscreen = { browserViewModel.exitFullscreen() }
   206→        )
   207→        // Don't render the rest of the UI while in fullscreen
   208→        return
   209→    }
   210→
   211→    // ═══════════════════════════════════════════════════════════════════
   212→    // Quick Switcher overlay
   213→    // ═══════════════════════════════════════════════════════════════════
   214→    if (uiState.showQuickSwitcher) {
   215→        QuickSwitcherOverlay(
   216→            query = uiState.quickSwitcherQuery,
   217→            tabs = uiState.filteredTabs,
   218→            activeTabId = uiState.activeTabId,
   219→            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   220→            onTabSelected = {
   221→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   222→                mainViewModel.onEvent(MainEvent.DismissAll)
   223→            },
   224→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   225→        )
   226→        return
   227→    }
   228→
   229→    // ═══════════════════════════════════════════════════════════════════
   230→    // Tab Overview (fullscreen grid)
   231→    // ═══════════════════════════════════════════════════════════════════
   232→    if (uiState.showTabOverview) {
   233→        TabOverviewScreen(
   234→            viewModel = hiltViewModel(),
   235→            onTabSelected = {
   236→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   237→                mainViewModel.onEvent(MainEvent.DismissAll)
   238→            },
   239→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   240→            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   241→        )
   242→        return
   243→    }
   244→
   245→    // ═══════════════════════════════════════════════════════════════════
   246→    // Main browser layout
   247→    // ═══════════════════════════════════════════════════════════════════
   248→    ModalNavigationDrawer(
   249→        drawerState = drawerState,
   250→        drawerContent = {
   251→            WorkspaceDrawerContent(
   252→                workspaces = uiState.workspaces,
   253→                activeWorkspaceId = uiState.activeWorkspaceId,
   254→                onWorkspaceSelected = { wsId ->
   255→                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   256→                    coroutineScope.launch { drawerState.close() }
   257→                },
   258→                onNavigateToBookmarks = {
   259→                    coroutineScope.launch { drawerState.close() }
   260→                    onNavigateToBookmarks()
   261→                },
   262→                onNavigateToSettings = {
   263→                    coroutineScope.launch { drawerState.close() }
   264→                    onNavigateToSettings()
   265→                },
   266→                onNavigateToHistory = {
   267→                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   268→                    coroutineScope.launch { drawerState.close() }
   269→                }
   270→            )
   271→        }
   272→    ) {
   273→        Box(
   274→            modifier = Modifier
   275→                .fillMaxSize()
   276→        ) {
   277→            Scaffold(
   278→                modifier = Modifier
   279→                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   280→                    .navigationBarsPadding(),
   281→                topBar = {
   282→                    MainTopBar(
   283→                        uiState = uiState,
   284→                        browserState = browserState,
   285→                        scrollBehavior = scrollBehavior,
   286→                        onUrlSubmit = { input ->
   287→                            val result = UrlNormalizer.normalize(
   288→                                input,
   289→                                uiState.settings.searchEngineUrl
   290→                            )
   291→                            browserViewModel.loadUrl(result.url)
   292→                        },
   293→                        onBack = { browserViewModel.goBack() },
   294→                        onForward = { browserViewModel.goForward() },
   295→                        onRefresh = {
   296→                            if (browserState.isLoading) browserViewModel.stopLoading()
   297→                            else browserViewModel.reload()
   298→                        },
   299→                        onBookmarkToggle = {
   300→                            val tab = activeTab ?: return@MainTopBar
   301→                            mainViewModel.onEvent(
   302→                                MainEvent.ToggleBookmark(
   303→                                    url = tab.url,
   304→                                    title = tab.title,
   305→                                    faviconUrl = tab.faviconUrl
   306→                                )
   307→                            )
   308→                        },
   309→                        onMenuClick = { showOverflowMenu = true },
   310→                        onDrawerClick = {
   311→                            coroutineScope.launch { drawerState.open() }
   312→                        }
   313→                    )
   314→                },
   315→                bottomBar = {
   316→                    if (uiState.tabs.isNotEmpty()) {
   317→                        TabStrip(
   318→                            tabs = uiState.tabs,
   319→                            activeTabId = uiState.activeTabId,
   320→                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   321→                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   322→                            onTabLongClick = {
   323→                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   324→                            },
   325→                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   326→                        )
   327→                    }
   328→                },
   329→                floatingActionButton = {
   330→                    ExtendedNewTabFab(
   331→                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   332→                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   333→                    )
   334→                },
   335→                snackbarHost = { SnackbarHost(snackbarHostState) },
   336→                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   337→            ) { paddingValues ->
   338→
   339→                // ── WebView content area ────────────────────────────
   340→                Box(
   341→                    modifier = Modifier
   342→                        .fillMaxSize()
   343→                        .padding(paddingValues)
   344→                ) {
   345→                    val currentWebView = webView
   346→                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   347→                        // key() forces AndroidView to recreate when switching tabs,
   348→                        // ensuring the correct WebView is attached to the composition tree.
   349→                        // The WebViewManager owns the WebView lifecycle; AndroidView only manages
   350→                        // view-hierarchy attachment. When the key changes, Compose disposes
   351→                        // the old AndroidView (removing the previous WebView from the hierarchy)
   352→                        // and creates a new one (attaching the new WebView).
   353→                        key(activeTab.id) {
   354→                            AndroidView(
   355→                                factory = { ctx ->
   356→                                    // Safety: remove from any stale parent before re-attaching
   357→                                    (currentWebView.parent as? android.view.ViewGroup)?.removeView(currentWebView)
   358→                                    currentWebView
   359→                                },
   360→                                modifier = Modifier.fillMaxSize()
   361→                            )
   362→                        }
   363→                    } else if (activeTab?.isHibernated == true) {
   364→                        HibernatedTabPlaceholder(
   365→                            tab = activeTab,
   366→                            onWake = {
   367→                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   368→                            }
   369→                        )
   370→                    } else {
   371→                        HomePlaceholder(
   372→                            onSearch = { query ->
   373→                                val result = UrlNormalizer.normalize(
   374→                                    query,
   375→                                    uiState.settings.searchEngineUrl
   376→                                )
   377→                                if (activeTab != null) {
   378→                                    browserViewModel.loadUrl(result.url)
   379→                                } else {
   380→                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   381→                                }
   382→                            },
   383→                            onNavigateToBookmarks = onNavigateToBookmarks
   384→                        )
   385→                    }
   386→
   387→                }
   388→            }
   389→        }
   390→    }
   391→
   392→    // ═══════════════════════════════════════════════════════════════════
   393→    // Overflow dropdown menu
   394→    // ═══════════════════════════════════════════════════════════════════
   395→    if (showOverflowMenu) {
   396→        OverflowMenuDialog(
   397→            onDismiss = { showOverflowMenu = false },
   398→            onTabOverview = {
   399→                showOverflowMenu = false
   400→                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   401→            },
   402→            onNewTab = {
   403→                showOverflowMenu = false
   404→                mainViewModel.onEvent(MainEvent.ShowAddTab)
   405→            },
   406→            onBookmarks = {
   407→                showOverflowMenu = false
   408→                onNavigateToBookmarks()
   409→            },
   410→            onHistory = {
   411→                showOverflowMenu = false
   412→                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   413→            },
   414→            onSettings = {
   415→                showOverflowMenu = false
   416→                onNavigateToSettings()
   417→            }
   418→        )
   419→    }
   420→
   421→    // ═══════════════════════════════════════════════════════════════════
   422→    // Bottom sheets
   423→    // ═══════════════════════════════════════════════════════════════════
   424→    if (uiState.showWorkspaceSwitcher) {
   425→        WorkspaceSwitcherSheet(
   426→            viewModel = hiltViewModel(),
   427→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   428→        )
   429→    }
   430→
   431→    if (uiState.showAddTabSheet) {
   432→        AddTabSheet(
   433→            viewModel = hiltViewModel(),
   434→            onTabCreated = { params ->
   435→                mainViewModel.onEvent(MainEvent.AddTab(params.url, params.customName))
   436→                mainViewModel.onEvent(MainEvent.DismissAll)
   437→            },
   438→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   439→        )
   440→    }
   441→
   442→    if (uiState.showRecentlyClosed) {
   443→        RecentlyClosedSheet(
   444→            closedTabs = emptyList(),
   445→            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   446→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   447→            onDismissItem = {},
   448→            onClearAll = {}
   449→        )
   450→    }
   451→}

   457→@OptIn(ExperimentalMaterial3Api::class)
   458→@Composable
   459→private fun MainTopBar(
   460→    uiState: MainUiState,
   461→    browserState: BrowserUiState,
   462→    scrollBehavior: TopAppBarScrollBehavior,
   463→    onUrlSubmit: (String) -> Unit,
   464→    onBack: () -> Unit,
   465→    onForward: () -> Unit,
   466→    onRefresh: () -> Unit,
   467→    onBookmarkToggle: () -> Unit,
   468→    onMenuClick: () -> Unit,
   469→    onDrawerClick: () -> Unit,
   470→) {
   471→    Column {
   472→        TopAppBar(

   474→            navigationIcon = {

   478→                        contentDescription = "Workspaces",
   479→                        tint = MaterialTheme.colorScheme.onSurface
   480→                    )

   485→                Surface(
   486→                    shape = RoundedCornerShape(12.dp),
   487→                    color = MaterialTheme.colorScheme.surfaceVariant,
   488→                    modifier = Modifier
   489→                        .padding(end = 4.dp)
   490→                        .clickable { /* Tab count chip – no action here */ }
   491→                ) {
   492→                    Row(
   493→                        verticalAlignment = Alignment.CenterVertically,
   494→                        modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp)

   498→                            contentDescription = "Tabs",
   499→                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
   500→                            modifier = Modifier.size(16.dp)
   501→                        )
   502→                        Spacer(Modifier.width(4.dp))
   503→                        Text(
   504→                            text = "$tabCount",
   505→                            style = MaterialTheme.typography.labelSmall,
   506→                            color = MaterialTheme.colorScheme.onSurfaceVariant
   507→                        )
   508→                    }
   509→                }
   510→            },
   511→            colors = TopAppBarDefaults.topAppBarColors(
   512→                containerColor = Color.Transparent
   513→            ),
   514→            scrollBehavior = scrollBehavior,
   515→            modifier = Modifier.statusBarsPadding()
   516→        )
   517→
   518→        // Omnibox below the TopAppBar
   519→        Omnibox(
   520→            url = browserState.currentUrl,
   521→            title = browserState.currentTitle,
   522→            isLoading = browserState.isLoading,
   523→            progress = browserState.loadingProgress,
   524→            sslState = browserState.sslState,
   525→            canGoBack = browserState.canGoBack,
   526→            canGoForward = browserState.canGoForward,
   527→            onUrlSubmit = onUrlSubmit,
   528→            onBack = onBack,
   529→            onForward = onForward,
   530→            onRefresh = onRefresh,
   531→            onBookmarkToggle = onBookmarkToggle,
   532→            isBookmarked = uiState.isBookmarked,
   533→            onMenuClick = onMenuClick,
   534→            modifier = Modifier.padding(horizontal = 4.dp)
   535→        )
   536→    }
   537→}

   543→@Composable
   544→private fun WorkspaceDrawerContent(
   545→    workspaces: List<Workspace>,
   546→    activeWorkspaceId: String?,
   547→    onWorkspaceSelected: (String) -> Unit,
   548→    onNavigateToBookmarks: () -> Unit,
   549→    onNavigateToSettings: () -> Unit,
   550→    onNavigateToHistory: () -> Unit,
   551→) {
   552→    ModalDrawerSheet(
   553→        drawerContainerColor = MaterialTheme.colorScheme.surface
   554→    ) {
   555→        // Header
   556→        Column(
   557→            modifier = Modifier.padding(horizontal = 16.dp, vertical = 12.dp)
   558→        ) {
   559→            Text(
