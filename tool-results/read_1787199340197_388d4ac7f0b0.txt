   105→   105→// ─────────────────────────────────────────────────────────────────────────────
   106→   106→// MainScreen – the primary browser screen that orchestrates the entire app.
   107→   107→// ─────────────────────────────────────────────────────────────────────────────
   108→   108→
   109→   109→/**
   110→   110→ * The top-level browser screen that hosts the [Omnibox], WebView content,
   111→   111→ * [TabStrip], and all overlay bottom-sheets / dialogs.
   112→   112→ *
   113→   113→ * This composable is the primary entry point after onboarding and is
   114→   114→ * responsible for orchestrating the [MainViewModel] (tab & workspace
   115→   115→ * management) with the [BrowserViewModel] (WebView rendering).
   116→   116→ *
   117→   117→ * @param initialUrl          An optional URL from a share-intent or deep link.
   118→   118→ * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   119→   119→ * @param onNavigateToSettings  Callback to navigate to the settings screen.
   120→   120→ */
   121→   121→@OptIn(ExperimentalMaterial3Api::class)
   122→   122→@SuppressLint("SetJavaScriptEnabled")
   123→   123→@Composable
   124→   124→fun MainScreen(
   125→   125→    initialUrl: String? = null,
   126→   126→    onNavigateToBookmarks: () -> Unit = {},
   127→   127→    onNavigateToSettings: () -> Unit = {},
   128→   128→    mainViewModel: MainViewModel = hiltViewModel(),
   129→   129→    browserViewModel: BrowserViewModel = hiltViewModel(),
   130→   130→) {
   131→   131→    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   132→   132→    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   133→   133→    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   134→   134→    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   135→   135→
   136→   136→    val coroutineScope = rememberCoroutineScope()
   137→   137→    val snackbarHostState = remember { SnackbarHostState() }
   138→   138→
   139→   139→    // ── Handle initial URL from share intent / deep link ────────────
   140→   140→    val initialUrlHandled = remember { mutableStateOf(false) }
   141→   141→    LaunchedEffect(initialUrl) {
   142→   142→        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   143→   143→            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   144→   144→            initialUrlHandled.value = true
   145→   145→        }
   146→   146→    }
   147→   147→
   148→   148→    // ── Track active tab changes and attach to browser ──────────────
   149→   149→    val activeTab = uiState.activeTab
   150→   150→    LaunchedEffect(activeTab?.id) {
   151→   151→        if (activeTab != null && !activeTab.isHibernated) {
   152→   152→            browserViewModel.attachTab(activeTab.id, activeTab)
   153→   153→        }
   154→   154→    }
   155→   155→
   156→   156→    // Refresh bookmark status when the active tab URL changes
   157→   157→    LaunchedEffect(activeTab?.url) {
   158→   158→        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   159→   159→    }
   160→   160→
   161→   161→    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   162→   162→    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   163→   163→
   164→   164→    // ── Drawer state for workspace rail (wide screens / all screens) ─
   165→   165→    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   166→   166→
   167→   167→    // ── Menu expansion state ────────────────────────────────────────
   168→   168→    var showOverflowMenu by remember { mutableStateOf(false) }
   169→   169→
   170→   170→    // ═══════════════════════════════════════════════════════════════════
   171→   171→    // Back-press handling: sheets > WebView.goBack > close tab
   172→   172→    // ═══════════════════════════════════════════════════════════════════
   173→   173→    BackHandler(enabled = true) {
   174→   174→        when {
   175→   175→            uiState.showQuickSwitcher ||
   176→   176→            uiState.showWorkspaceSwitcher ||
   177→   177→            uiState.showAddTabSheet ||
   178→   178→            uiState.showRecentlyClosed -> {
   179→   179→                mainViewModel.onEvent(MainEvent.DismissAll)
   180→   180→            }
   181→   181→            uiState.showTabOverview -> {
   182→   182→                mainViewModel.onEvent(MainEvent.DismissAll)
   183→   183→            }
   184→   184→            showOverflowMenu -> {
   185→   185→                showOverflowMenu = false
   186→   186→            }
   187→   187→            browserState.canGoBack -> {
   188→   188→                browserViewModel.goBack()
   189→   189→            }
   190→   190→            activeTab != null -> {
   191→   191→                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   192→   192→            }
   193→   193→        }
   194→   194→    }
   195→   195→
   196→   196→    // ═══════════════════════════════════════════════════════════════════
   197→   197→    // Fullscreen video overlay
   198→   198→    // ═══════════════════════════════════════════════════════════════════
   199→   199→    if (customView != null) {
   200→   200→        FullscreenVideoOverlay(
   201→   201→            customView = customView,
   202→   202→            onExitFullscreen = { browserViewModel.exitFullscreen() }
   203→   203→        )
   204→   204→        // Don't render the rest of the UI while in fullscreen
   205→   205→        return
   206→   206→    }
   207→   207→
   208→   208→    // ═══════════════════════════════════════════════════════════════════
   209→   209→    // Quick Switcher overlay
   210→   210→    // ═══════════════════════════════════════════════════════════════════
   211→   211→    if (uiState.showQuickSwitcher) {
   212→   212→        QuickSwitcherOverlay(
   213→   213→            query = uiState.quickSwitcherQuery,
   214→   214→            tabs = uiState.filteredTabs,
   215→   215→            activeTabId = uiState.activeTabId,
   216→   216→            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   217→   217→            onTabSelected = {
   218→   218→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   219→   219→                mainViewModel.onEvent(MainEvent.DismissAll)
   220→   220→            },
   221→   221→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   222→   222→        )
   223→   223→        return
   224→   224→    }
   225→   225→
   226→   226→    // ═══════════════════════════════════════════════════════════════════
   227→   227→    // Tab Overview (fullscreen grid)
   228→   228→    // ═══════════════════════════════════════════════════════════════════
   229→   229→    if (uiState.showTabOverview) {
   230→   230→        TabOverviewScreen(
   231→   231→            viewModel = hiltViewModel(),
   232→   232→            onTabSelected = {
   233→   233→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   234→   234→                mainViewModel.onEvent(MainEvent.DismissAll)
   235→   235→            },
   236→   236→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   237→   237→            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   238→   238→        )
   239→   239→        return
   240→   240→    }
   241→   241→
   242→   242→    // ═══════════════════════════════════════════════════════════════════
   243→   243→    // Main browser layout
   244→   244→    // ═══════════════════════════════════════════════════════════════════
   245→   245→    ModalNavigationDrawer(
   246→   246→        drawerState = drawerState,
   247→   247→        drawerContent = {
   248→   248→            WorkspaceDrawerContent(
   249→   249→                workspaces = uiState.workspaces,
   250→   250→                activeWorkspaceId = uiState.activeWorkspaceId,
   251→   251→                onWorkspaceSelected = { wsId ->
   252→   252→                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   253→   253→                    coroutineScope.launch { drawerState.close() }
   254→   254→                },
   255→   255→                onNavigateToBookmarks = {
   256→   256→                    coroutineScope.launch { drawerState.close() }
   257→   257→                    onNavigateToBookmarks()
   258→   258→                },
   259→   259→                onNavigateToSettings = {
   260→   260→                    coroutineScope.launch { drawerState.close() }
   261→   261→                    onNavigateToSettings()
   262→   262→                },
   263→   263→                onNavigateToHistory = {
   264→   264→                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   265→   265→                    coroutineScope.launch { drawerState.close() }
   266→   266→                }
   267→   267→            )
   268→   268→        }
   269→   269→    ) {
   270→   270→        Box(
   271→   271→            modifier = Modifier
   272→   272→                .fillMaxSize()
   273→   273→        ) {
   274→   274→            Scaffold(
   275→   275→                modifier = Modifier
   276→   276→                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   277→   277→                    .navigationBarsPadding(),
   278→   278→                topBar = {
   279→   279→                    MainTopBar(
   280→   280→                        uiState = uiState,
   281→   281→                        browserState = browserState,
   282→   282→                        scrollBehavior = scrollBehavior,
   283→   283→                        onUrlSubmit = { input ->
   284→   284→                            val result = UrlNormalizer.normalize(
   285→   285→                                input,
   286→   286→                                uiState.settings.searchEngineUrl
   287→   287→                            )
   288→   288→                            browserViewModel.loadUrl(result.url)
   289→   289→                        },
   290→   290→                        onBack = { browserViewModel.goBack() },
   291→   291→                        onForward = { browserViewModel.goForward() },
   292→   292→                        onRefresh = {
   293→   293→                            if (browserState.isLoading) browserViewModel.stopLoading()
   294→   294→                            else browserViewModel.reload()
   295→   295→                        },
   296→   296→                        onBookmarkToggle = {
   297→   297→                            val tab = activeTab ?: return@MainTopBar
   298→   298→                            mainViewModel.onEvent(
   299→   299→                                MainEvent.ToggleBookmark(
   300→   300→                                    url = tab.url,
   301→   301→                                    title = tab.title,
   302→   302→                                    faviconUrl = tab.faviconUrl
   303→   303→                                )
   304→   304→                            )
   305→   305→                        },
   306→   306→                        onMenuClick = { showOverflowMenu = true },
   307→   307→                        onDrawerClick = {
   308→   308→                            coroutineScope.launch { drawerState.open() }
   309→   309→                        }
   310→   310→                    )
   311→   311→                },
   312→   312→                bottomBar = {
   313→   313→                    if (uiState.tabs.isNotEmpty()) {
   314→   314→                        TabStrip(
   315→   315→                            tabs = uiState.tabs,
   316→   316→                            activeTabId = uiState.activeTabId,
   317→   317→                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   318→   318→                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   319→   319→                            onTabLongClick = {
   320→   320→                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   321→   321→                            },
   322→   322→                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   323→   323→                        )
   324→   324→                    }
   325→   325→                },
   326→   326→                floatingActionButton = {
   327→   327→                    ExtendedNewTabFab(
   328→   328→                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   329→   329→                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   330→   330→                    )
   331→   331→                },
   332→   332→                snackbarHost = { SnackbarHost(snackbarHostState) },
   333→   333→                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   334→   334→            ) { paddingValues ->
   335→   335→
   336→   336→                // ── WebView content area ────────────────────────────
   337→   337→                Box(
   338→   338→                    modifier = Modifier
   339→   339→                        .fillMaxSize()
   340→   340→                        .padding(paddingValues)
   341→   341→                ) {
   342→   342→                    val currentWebView = webView
   343→   343→                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   344→   344→                        AndroidView(
   345→   345→                            factory = { currentWebView },
   346→   346→                            modifier = Modifier.fillMaxSize()
   347→   347→                        )
   348→   348→                    } else if (activeTab?.isHibernated == true) {
   349→   349→                        HibernatedTabPlaceholder(
   350→   350→                            tab = activeTab,
   351→   351→                            onWake = {
   352→   352→                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   353→   353→                            }
   354→   354→                        )
   355→   355→                    } else {
   356→   356→                        HomePlaceholder(
   357→   357→                            onSearch = { query ->
   358→   358→                                val result = UrlNormalizer.normalize(
   359→   359→                                    query,
   360→   360→                                    uiState.settings.searchEngineUrl
   361→   361→                                )
   362→   362→                                if (activeTab != null) {
   363→   363→                                    browserViewModel.loadUrl(result.url)
   364→   364→                                } else {
   365→   365→                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   366→   366→                                }
   367→   367→                            },
   368→   368→                            onNavigateToBookmarks = onNavigateToBookmarks
   369→   369→                        )
   370→   370→                    }
   371→   371→
   372→   372→                }
   373→   373→            }
   374→   374→        }
   375→   375→    }
   376→   376→
   377→   377→    // ═══════════════════════════════════════════════════════════════════
   378→   378→    // Overflow dropdown menu
   379→   379→    // ═══════════════════════════════════════════════════════════════════
   380→   380→    if (showOverflowMenu) {
   381→   381→        OverflowMenuDialog(
   382→   382→            onDismiss = { showOverflowMenu = false },
   383→   383→            onTabOverview = {
   384→   384→                showOverflowMenu = false
   385→   385→                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   386→   386→            },
   387→   387→            onNewTab = {
   388→   388→                showOverflowMenu = false
   389→   389→                mainViewModel.onEvent(MainEvent.ShowAddTab)
   390→   390→            },
   391→   391→            onBookmarks = {
   392→   392→                showOverflowMenu = false
   393→   393→                onNavigateToBookmarks()
   394→   394→            },
   395→   395→            onHistory = {
   396→   396→                showOverflowMenu = false
   397→   397→                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   398→   398→            },
   399→   399→            onSettings = {
   400→   400→                showOverflowMenu = false
   401→   401→                onNavigateToSettings()
   402→   402→            }
   403→   403→        )
   404→   404→    }
   405→   405→
   406→   406→    // ═══════════════════════════════════════════════════════════════════
   407→   407→    // Bottom sheets
   408→   408→    // ═══════════════════════════════════════════════════════════════════
   409→   409→    if (uiState.showWorkspaceSwitcher) {
   410→   410→        WorkspaceSwitcherSheet(
   411→   411→            viewModel = hiltViewModel(),
   412→   412→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   413→   413→        )
   414→   414→    }
   415→   415→
   416→   416→    if (uiState.showAddTabSheet) {
   417→   417→        AddTabSheet(
   418→   418→            viewModel = hiltViewModel(),
   419→   419→            onTabCreated = { params ->
   420→   420→                mainViewModel.onEvent(MainEvent.AddTab(params.url, params.customName))
   421→   421→                mainViewModel.onEvent(MainEvent.DismissAll)
   422→   422→            },
   423→   423→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   424→   424→        )
   425→   425→    }
   426→   426→
   427→   427→    if (uiState.showRecentlyClosed) {
   428→   428→        RecentlyClosedSheet(
   429→   429→            closedTabs = emptyList(),
   430→   430→            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   431→   431→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   432→   432→            onDismissItem = {},
   433→   433→            onClearAll = {}
   434→   434→        )
   435→   435→    }
   436→   436→}
   437→   437→
   438→   438→// ─────────────────────────────────────────────────────────────────────────────
   439→   439→// TopAppBar with Omnibox

   442→   442→@OptIn(ExperimentalMaterial3Api::class)
   443→   443→@Composable
   444→   444→private fun MainTopBar(
   445→   445→    uiState: MainUiState,
   446→   446→    browserState: BrowserUiState,
   447→   447→    scrollBehavior: TopAppBarScrollBehavior,
   448→   448→    onUrlSubmit: (String) -> Unit,
   449→   449→    onBack: () -> Unit,
   450→   450→    onForward: () -> Unit,
   451→   451→    onRefresh: () -> Unit,
   452→   452→    onBookmarkToggle: () -> Unit,
   453→   453→    onMenuClick: () -> Unit,
   454→   454→    onDrawerClick: () -> Unit,
   455→   455→) {
   456→   456→    Column {
   457→   457→        TopAppBar(
