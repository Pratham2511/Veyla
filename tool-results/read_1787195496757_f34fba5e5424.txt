   572→@OptIn(ExperimentalMaterial3Api::class)
   573→@Composable
   574→private fun SearchEngineDropdown(
   575→    engines: List<Pair<String, String>>,
   576→    currentEngineName: String,
   577→    onEngineSelected: (String) -> Unit,
   578→) {
   579→    var expanded by remember { mutableStateOf(false) }
   580→
   581→    Column(modifier = Modifier.padding(horizontal = 16.dp, vertical = 4.dp)) {
   582→        Text(
   583→            text = "Default search engine",
   584→            style = MaterialTheme.typography.bodyMedium,
   585→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   586→            modifier = Modifier.padding(bottom = 8.dp)
   587→        )
   588→
   589→        ExposedDropdownMenuBox(
   590→            expanded = expanded,
   591→            onExpandedChange = { expanded = it }
   592→        ) {
   593→            OutlinedTextField(
   594→                value = currentEngineName,
   595→                onValueChange = {},
   596→                readOnly = true,
   597→                label = { Text("Search engine") },
   598→                trailingIcon = {
   599→                    ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
   600→                },
   601→                leadingIcon = {
   602→                    Icon(Icons.Default.Search, contentDescription = null)
   603→                },
   604→                modifier = Modifier
   605→                    .menuAnchor(MenuAnchorType.PrimaryNotEditable)
   606→                    .fillMaxWidth()
   607→            )
   608→
   609→            ExposedDropdownMenu(
   610→                expanded = expanded,
   611→                onDismissRequest = { expanded = false }
   612→            ) {
   613→                engines.forEach { (name, url) ->
   614→                    androidx.compose.material3.DropdownMenuItem(
   615→                        text = { Text(name) },
   616→                        onClick = {
   617→                            onEngineSelected(url)
   618→                            expanded = false
   619→                        },
   620→                        trailingIcon = {
   621→                            if (name == currentEngineName) {
   622→                                Text(
   623→                                    "\u2713", // checkmark
   624→                                    color = MaterialTheme.colorScheme.primary
   625→                                )
   626→                            }
   627→                        }
   628→                    )
   629→                }
   630→            }
   631→        }
   632→    }
   633→}

   635→@Composable
   636→private fun SessionListItem(
   637→    session: com.pratham.webhub.domain.model.SessionSnapshot,
   638→    onRestore: () -> Unit,
   639→    onDelete: () -> Unit,
   640→) {
   641→    val dateFormatted = remember(session.createdAt) {
