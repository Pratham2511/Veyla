   100→   100→// ─────────────────────────────────────────────────────────────────────────────
   101→   101→// MainScreen – the primary browser screen that orchestrates the entire app.
   102→   102→// ─────────────────────────────────────────────────────────────────────────────
   103→   103→
   104→   104→/**
   105→   105→ * The top-level browser screen that hosts the [Omnibox], WebView content,
   106→   106→ * [TabStrip], and all overlay bottom-sheets / dialogs.
   107→   107→ *
   108→   108→ * This composable is the primary entry point after onboarding and is
   109→   109→ * responsible for orchestrating the [MainViewModel] (tab & workspace
   110→   110→ * management) with the [BrowserViewModel] (WebView rendering).
   111→   111→ *
   112→   112→ * @param initialUrl          An optional URL from a share-intent or deep link.
   113→   113→ * @param onNavigateToBookmarks Callback to navigate to the bookmarks screen.
   114→   114→ * @param onNavigateToSettings  Callback to navigate to the settings screen.
   115→   115→ */
   116→   116→@OptIn(ExperimentalMaterial3Api::class)
   117→   117→@SuppressLint("SetJavaScriptEnabled")
   118→   118→@Composable
   119→   119→fun MainScreen(
   120→   120→    initialUrl: String? = null,
   121→   121→    onNavigateToBookmarks: () -> Unit = {},
   122→   122→    onNavigateToSettings: () -> Unit = {},
   123→   123→    mainViewModel: MainViewModel = hiltViewModel(),
   124→   124→    browserViewModel: BrowserViewModel = hiltViewModel(),
   125→   125→) {
   126→   126→    val uiState by mainViewModel.state.collectAsStateWithLifecycle()
   127→   127→    val browserState by browserViewModel.state.collectAsStateWithLifecycle()
   128→   128→    val webView by browserViewModel.webView.collectAsStateWithLifecycle()
   129→   129→    val customView by browserViewModel.customView.collectAsStateWithLifecycle()
   130→   130→
   131→   131→    val coroutineScope = rememberCoroutineScope()
   132→   132→    val snackbarHostState = remember { SnackbarHostState() }
   133→   133→
   134→   134→    // ── Handle initial URL from share intent / deep link ────────────
   135→   135→    val initialUrlHandled = remember { mutableStateOf(false) }
   136→   136→    LaunchedEffect(initialUrl) {
   137→   137→        if (!initialUrlHandled.value && !initialUrl.isNullOrBlank()) {
   138→   138→            mainViewModel.onEvent(MainEvent.AddTab(initialUrl, null))
   139→   139→            initialUrlHandled.value = true
   140→   140→        }
   141→   141→    }
   142→   142→
   143→   143→    // ── Track active tab changes and attach to browser ──────────────
   144→   144→    val activeTab = uiState.activeTab
   145→   145→    LaunchedEffect(activeTab?.id) {
   146→   146→        if (activeTab != null && !activeTab.isHibernated) {
   147→   147→            browserViewModel.attachTab(activeTab.id, activeTab)
   148→   148→        }
   149→   149→    }
   150→   150→
   151→   151→    // Refresh bookmark status when the active tab URL changes
   152→   152→    LaunchedEffect(activeTab?.url) {
   153→   153→        activeTab?.let { mainViewModel.refreshBookmarkStatus(it.url) }
   154→   154→    }
   155→   155→
   156→   156→    // ── Scroll behaviour for TopAppBar ──────────────────────────────
   157→   157→    val scrollBehavior = TopAppBarDefaults.pinnedScrollBehavior(rememberTopAppBarState())
   158→   158→
   159→   159→    // ── Drawer state for workspace rail (wide screens / all screens) ─
   160→   160→    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
   161→   161→
   162→   162→    // ── Menu expansion state ────────────────────────────────────────
   163→   163→    var showOverflowMenu by remember { mutableStateOf(false) }
   164→   164→
   165→   165→    // ═══════════════════════════════════════════════════════════════════
   166→   166→    // Back-press handling: sheets > WebView.goBack > close tab
   167→   167→    // ═══════════════════════════════════════════════════════════════════
   168→   168→    BackHandler(enabled = true) {
   169→   169→        when {
   170→   170→            uiState.showQuickSwitcher ||
   171→   171→            uiState.showWorkspaceSwitcher ||
   172→   172→            uiState.showAddTabSheet ||
   173→   173→            uiState.showRecentlyClosed -> {
   174→   174→                mainViewModel.onEvent(MainEvent.DismissAll)
   175→   175→            }
   176→   176→            uiState.showTabOverview -> {
   177→   177→                mainViewModel.onEvent(MainEvent.DismissAll)
   178→   178→            }
   179→   179→            showOverflowMenu -> {
   180→   180→                showOverflowMenu = false
   181→   181→            }
   182→   182→            browserState.canGoBack -> {
   183→   183→                browserViewModel.goBack()
   184→   184→            }
   185→   185→            activeTab != null -> {
   186→   186→                mainViewModel.onEvent(MainEvent.CloseTab(activeTab.id))
   187→   187→            }
   188→   188→        }
   189→   189→    }
   190→   190→
   191→   191→    // ═══════════════════════════════════════════════════════════════════
   192→   192→    // Fullscreen video overlay
   193→   193→    // ═══════════════════════════════════════════════════════════════════
   194→   194→    if (customView != null) {
   195→   195→        FullscreenVideoOverlay(
   196→   196→            customView = customView,
   197→   197→            onExitFullscreen = { browserViewModel.exitFullscreen() }
   198→   198→        )
   199→   199→        // Don't render the rest of the UI while in fullscreen
   200→   200→        return
   201→   201→    }
   202→   202→
   203→   203→    // ═══════════════════════════════════════════════════════════════════
   204→   204→    // Quick Switcher overlay
   205→   205→    // ═══════════════════════════════════════════════════════════════════
   206→   206→    if (uiState.showQuickSwitcher) {
   207→   207→        QuickSwitcherOverlay(
   208→   208→            query = uiState.quickSwitcherQuery,
   209→   209→            tabs = uiState.filteredTabs,
   210→   210→            activeTabId = uiState.activeTabId,
   211→   211→            onQueryChanged = { mainViewModel.onEvent(MainEvent.UpdateQuickSwitcherQuery(it)) },
   212→   212→            onTabSelected = {
   213→   213→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   214→   214→                mainViewModel.onEvent(MainEvent.DismissAll)
   215→   215→            },
   216→   216→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   217→   217→        )
   218→   218→        return
   219→   219→    }
   220→   220→
   221→   221→    // ═══════════════════════════════════════════════════════════════════
   222→   222→    // Tab Overview (fullscreen grid)
   223→   223→    // ═══════════════════════════════════════════════════════════════════
   224→   224→    if (uiState.showTabOverview) {
   225→   225→        TabOverviewScreen(
   226→   226→            onTabSelected = {
   227→   227→                mainViewModel.onEvent(MainEvent.SelectTab(it))
   228→   228→                mainViewModel.onEvent(MainEvent.DismissAll)
   229→   229→            },
   230→   230→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) },
   231→   231→            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   232→   232→        )
   233→   233→        return
   234→   234→    }
   235→   235→
   236→   236→    // ═══════════════════════════════════════════════════════════════════
   237→   237→    // Main browser layout
   238→   238→    // ═══════════════════════════════════════════════════════════════════
   239→   239→    ModalNavigationDrawer(
   240→   240→        drawerState = drawerState,
   241→   241→        drawerContent = {
   242→   242→            WorkspaceDrawerContent(
   243→   243→                workspaces = uiState.workspaces,
   244→   244→                activeWorkspaceId = uiState.activeWorkspaceId,
   245→   245→                onWorkspaceSelected = { wsId ->
   246→   246→                    mainViewModel.onEvent(MainEvent.SwitchWorkspace(wsId))
   247→   247→                    coroutineScope.launch { drawerState.close() }
   248→   248→                },
   249→   249→                onNavigateToBookmarks = {
   250→   250→                    coroutineScope.launch { drawerState.close() }
   251→   251→                    onNavigateToBookmarks()
   252→   252→                },
   253→   253→                onNavigateToSettings = {
   254→   254→                    coroutineScope.launch { drawerState.close() }
   255→   255→                    onNavigateToSettings()
   256→   256→                },
   257→   257→                onNavigateToHistory = {
   258→   258→                    mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   259→   259→                    coroutineScope.launch { drawerState.close() }
   260→   260→                }
   261→   261→            )
   262→   262→        }
   263→   263→    ) {
   264→   264→        Box(
   265→   265→            modifier = Modifier
   266→   266→                .fillMaxSize()
   267→   267→        ) {
   268→   268→            Scaffold(
   269→   269→                modifier = Modifier
   270→   270→                    .nestedScroll(scrollBehavior.nestedScrollConnection)
   271→   271→                    .navigationBarsPadding(),
   272→   272→                topBar = {
   273→   273→                    MainTopBar(
   274→   274→                        uiState = uiState,
   275→   275→                        browserState = browserState,
   276→   276→                        scrollBehavior = scrollBehavior,
   277→   277→                        onUrlSubmit = { input ->
   278→   278→                            val result = UrlNormalizer.normalize(
   279→   279→                                input,
   280→   280→                                uiState.settings.searchEngineUrl
   281→   281→                            )
   282→   282→                            browserViewModel.loadUrl(result.url)
   283→   283→                        },
   284→   284→                        onBack = { browserViewModel.goBack() },
   285→   285→                        onForward = { browserViewModel.goForward() },
   286→   286→                        onRefresh = {
   287→   287→                            if (browserState.isLoading) browserViewModel.stopLoading()
   288→   288→                            else browserViewModel.reload()
   289→   289→                        },
   290→   290→                        onBookmarkToggle = {
   291→   291→                            val tab = activeTab ?: return@MainTopBar
   292→   292→                            mainViewModel.onEvent(
   293→   293→                                MainEvent.ToggleBookmark(
   294→   294→                                    url = tab.url,
   295→   295→                                    title = tab.title,
   296→   296→                                    faviconUrl = tab.faviconUrl
   297→   297→                                )
   298→   298→                            )
   299→   299→                        },
   300→   300→                        onMenuClick = { showOverflowMenu = true },
   301→   301→                        onDrawerClick = {
   302→   302→                            coroutineScope.launch { drawerState.open() }
   303→   303→                        }
   304→   304→                    )
   305→   305→                },
   306→   306→                bottomBar = {
   307→   307→                    if (uiState.tabs.isNotEmpty()) {
   308→   308→                        TabStrip(
   309→   309→                            tabs = uiState.tabs,
   310→   310→                            activeTabId = uiState.activeTabId,
   311→   311→                            onTabSelected = { mainViewModel.onEvent(MainEvent.SelectTab(it)) },
   312→   312→                            onTabClosed = { mainViewModel.onEvent(MainEvent.CloseTab(it)) },
   313→   313→                            onTabLongClick = {
   314→   314→                                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   315→   315→                            },
   316→   316→                            onAddTab = { mainViewModel.onEvent(MainEvent.ShowAddTab) }
   317→   317→                        )
   318→   318→                    }
   319→   319→                },
   320→   320→                floatingActionButton = {
   321→   321→                    ExtendedNewTabFab(
   322→   322→                        onClick = { mainViewModel.onEvent(MainEvent.ShowAddTab) },
   323→   323→                        onLongClick = { mainViewModel.onEvent(MainEvent.ShowQuickSwitcher) }
   324→   324→                    )
   325→   325→                },
   326→   326→                snackbarHost = { SnackbarHost(snackbarHostState) },
   327→   327→                contentWindowInsets = ScaffoldDefaults.contentWindowInsets
   328→   328→            ) { paddingValues ->
   329→   329→
   330→   330→                // ── WebView content area ────────────────────────────
   331→   331→                Box(
   332→   332→                    modifier = Modifier
   333→   333→                        .fillMaxSize()
   334→   334→                        .padding(paddingValues)
   335→   335→                ) {
   336→   336→                    val currentWebView = webView
   337→   337→                    if (currentWebView != null && activeTab != null && !activeTab.isHibernated) {
   338→   338→                        AndroidView(
   339→   339→                            factory = { currentWebView },
   340→   340→                            modifier = Modifier.fillMaxSize()
   341→   341→                        )
   342→   342→                    } else if (activeTab?.isHibernated == true) {
   343→   343→                        HibernatedTabPlaceholder(
   344→   344→                            tab = activeTab,
   345→   345→                            onWake = {
   346→   346→                                mainViewModel.onEvent(MainEvent.SelectTab(activeTab.id))
   347→   347→                            }
   348→   348→                        )
   349→   349→                    } else {
   350→   350→                        HomePlaceholder(
   351→   351→                            onSearch = { query ->
   352→   352→                                val result = UrlNormalizer.normalize(
   353→   353→                                    query,
   354→   354→                                    uiState.settings.searchEngineUrl
   355→   355→                                )
   356→   356→                                if (activeTab != null) {
   357→   357→                                    browserViewModel.loadUrl(result.url)
   358→   358→                                } else {
   359→   359→                                    mainViewModel.onEvent(MainEvent.AddTab(result.url, null))
   360→   360→                                }
   361→   361→                            },
   362→   362→                            onNavigateToBookmarks = onNavigateToBookmarks
   363→   363→                        )
   364→   364→                    }
   365→   365→
   366→   366→                }
   367→   367→            }
   368→   368→        }
   369→   369→    }
   370→   370→
   371→   371→    // ═══════════════════════════════════════════════════════════════════
   372→   372→    // Overflow dropdown menu
   373→   373→    // ═══════════════════════════════════════════════════════════════════
   374→   374→    if (showOverflowMenu) {
   375→   375→        OverflowMenuDialog(
   376→   376→            onDismiss = { showOverflowMenu = false },
   377→   377→            onTabOverview = {
   378→   378→                showOverflowMenu = false
   379→   379→                mainViewModel.onEvent(MainEvent.ShowTabOverview)
   380→   380→            },
   381→   381→            onNewTab = {
   382→   382→                showOverflowMenu = false
   383→   383→                mainViewModel.onEvent(MainEvent.ShowAddTab)
   384→   384→            },
   385→   385→            onBookmarks = {
   386→   386→                showOverflowMenu = false
   387→   387→                onNavigateToBookmarks()
   388→   388→            },
   389→   389→            onHistory = {
   390→   390→                showOverflowMenu = false
   391→   391→                mainViewModel.onEvent(MainEvent.ShowRecentlyClosed)
   392→   392→            },
   393→   393→            onSettings = {
   394→   394→                showOverflowMenu = false
   395→   395→                onNavigateToSettings()
   396→   396→            }
   397→   397→        )
   398→   398→    }
   399→   399→
   400→   400→    // ═══════════════════════════════════════════════════════════════════
   401→   401→    // Bottom sheets
   402→   402→    // ═══════════════════════════════════════════════════════════════════
   403→   403→    if (uiState.showWorkspaceSwitcher) {
   404→   404→        WorkspaceSwitcherSheet(
   405→   405→            viewModel = hiltViewModel(),
   406→   406→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   407→   407→        )
   408→   408→    }
   409→   409→
   410→   410→    if (uiState.showAddTabSheet) {
   411→   411→        AddTabSheet(
   412→   412→            viewModel = hiltViewModel(),
   413→   413→            onTabCreated = { url, name ->
   414→   414→                mainViewModel.onEvent(MainEvent.AddTab(url, name))
   415→   415→                mainViewModel.onEvent(MainEvent.DismissAll)
   416→   416→            },
   417→   417→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   418→   418→        )
   419→   419→    }
   420→   420→
   421→   421→    if (uiState.showRecentlyClosed) {
   422→   422→        RecentlyClosedSheet(
   423→   423→            onRestore = { mainViewModel.onEvent(MainEvent.RestoreTab(it)) },
   424→   424→            onDismiss = { mainViewModel.onEvent(MainEvent.DismissAll) }
   425→   425→        )
   426→   426→    }
   427→   427→}
   428→   428→

   433→   433→@OptIn(ExperimentalMaterial3Api::class)
   434→   434→@Composable
   435→   435→private fun MainTopBar(
   436→   436→    uiState: MainUiState,
   437→   437→    browserState: BrowserUiState,
   438→   438→    scrollBehavior: TopAppBarScrollBehavior,
   439→   439→    onUrlSubmit: (String) -> Unit,
   440→   440→    onBack: () -> Unit,
   441→   441→    onForward: () -> Unit,
   442→   442→    onRefresh: () -> Unit,
   443→   443→    onBookmarkToggle: () -> Unit,
   444→   444→    onMenuClick: () -> Unit,
   445→   445→    onDrawerClick: () -> Unit,
   446→   446→) {
   447→   447→    Column {
   448→   448→        TopAppBar(
   449→   449→            title = {},
   450→   450→            navigationIcon = {
   451→   451→                IconButton(onClick = onDrawerClick) {
   452→   452→                    Icon(
   453→   453→                        imageVector = Icons.Default.Folder,
   454→   454→                        contentDescription = "Workspaces",
   455→   455→                        tint = MaterialTheme.colorScheme.onSurface
   456→   456→                    )
   457→   457→                }
   458→   458→            },
   459→   459→            actions = {
   460→   460→                val tabCount = uiState.tabs.size
   461→   461→                Surface(
