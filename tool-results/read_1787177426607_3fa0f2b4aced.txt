   574→@OptIn(ExperimentalMaterial3Api::class)
   575→@Composable
   576→private fun SearchEngineDropdown(
   577→    engines: List<Pair<String, String>>,
   578→    currentEngineName: String,
   579→    onEngineSelected: (String) -> Unit,
   580→) {
   581→    var expanded by remember { mutableStateOf(false) }
   582→
   583→    Column(modifier = Modifier.padding(horizontal = 16.dp, vertical = 4.dp)) {
   584→        Text(
   585→            text = "Default search engine",
   586→            style = MaterialTheme.typography.bodyMedium,
   587→            color = MaterialTheme.colorScheme.onSurfaceVariant,
   588→            modifier = Modifier.padding(bottom = 8.dp)
   589→        )
   590→
   591→        ExposedDropdownMenuBox(
   592→            expanded = expanded,
   593→            onExpandedChange = { expanded = it }
   594→        ) {
   595→            OutlinedTextField(
   596→                value = currentEngineName,
   597→                onValueChange = {},
   598→                readOnly = true,
   599→                label = { Text("Search engine") },
   600→                trailingIcon = {
   601→                    ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
   602→                },
   603→                leadingIcon = {
   604→                    Icon(Icons.Default.Search, contentDescription = null)
   605→                },
   606→                modifier = Modifier
   607→                    .menuAnchor(MenuAnchorType.PrimaryNotEditable)
   608→                    .fillMaxWidth()
   609→            )
   610→
   611→            ExposedDropdownMenu(
   612→                expanded = expanded,
   613→                onDismissRequest = { expanded = false }
   614→            ) {
   615→                engines.forEach { (name, url) ->
   616→                    androidx.compose.material3.DropdownMenuItem(
   617→                        text = { Text(name) },
   618→                        onClick = {
   619→                            onEngineSelected(url)
   620→                            expanded = false
   621→                        },
   622→                        trailingIcon = {
   623→                            if (name == currentEngineName) {
   624→                                Text(
   625→                                    "\u2713", // checkmark
   626→                                    color = MaterialTheme.colorScheme.primary
   627→                                )
   628→                            }
   629→                        }
   630→                    )
   631→                }
   632→            }
   633→        }
   634→    }

   637→@Composable
   638→private fun SessionListItem(
   639→    session: com.pratham.webhub.domain.model.SessionSnapshot,
   640→    onRestore: () -> Unit,
   641→    onDelete: () -> Unit,
   642→) {
   643→    val dateFormatted = remember(session.createdAt) {
   644→        val sdf = SimpleDateFormat("MMM dd, yyyy \u2022 hh:mm a", Locale.getDefault())
   645→        sdf.format(Date(session.createdAt))
   646→    }
   647→
   648→    ListItem(
   649→        headlineContent = {
   650→            Text(
   651→                session.name,
   652→                maxLines = 1,
   653→                overflow = TextOverflow.Ellipsis
   654→            )
   655→        },
   656→        supportingContent = { Text(dateFormatted) },
   657→        leadingContent = {
   658→            Icon(
   659→                imageVector = Icons.Default.History,
   660→                contentDescription = null
